-- Combinar las tablas pedidos, clientes y productos
SELECT
	pedidos.id,
	clientes.nombre AS cliente,
	productos.nombre AS producto,
	pedidos.cantidad,
	pedidos.fecha
FROM pedidos
JOIN clientes ON pedidos.cliente_id = clientes.id
JOIN productos ON pedidos.producto_id = productos.id;
