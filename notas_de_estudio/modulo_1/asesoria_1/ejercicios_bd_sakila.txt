Consulta solo titles de los films
SELECT title FROM film;

Consulta películas por distintas duraciones, menos de 90 mins y más de 160 mins.
SELECT title, length FROM film WHERE length < 90 OR length > 160;

Ordena las películas por su duración y obten el top 10 más largos y el top 10 más cortas.
SELECT title, length FROM film ORDER BY length DESC LIMIT 10;
SELECT title, length FROM film ORDER BY length LIMIT 10;
