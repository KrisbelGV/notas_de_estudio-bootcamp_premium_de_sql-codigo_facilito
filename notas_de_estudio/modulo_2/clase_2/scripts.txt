/* Eliminar registros de la tabla con DELETE dentro de una transaccion y deshacer cambios con rollback
DELETE FROM sin emplear WHERE Solo funciona con la configuracion SQL_SAFE_UPDATES en 0 
Los cambios se guardan en el archivo log, ocupando memoria*/
SET SQL_SAFE_UPDATES = 0;
START TRANSACTION; -- Iniciar transaccion
	DELETE FROM prueba_truncate;
ROLLBACK; -- Deshacer cambios

/* Eliminar registros de la tabla con TRUNCATE dentro de una transaccion e intentar deshacer cambios con rollback
No deja logs, por lo que ROLLACK no podra restituir los registros */
START TRANSACTION;
	TRUNCATE prueba_truncate;
ROLLBACK;

/* Agregar fecha_registro a clientes con valor por defecto
TIMESTAMP es un dato de tipo fecha en formato 'YY-MM-DD HH:MM:SS' desde la zona horaria del servidor
DEFAULT determina un valor por defecto
CURRENT_TIMESTAMP genera la fecha y hora actual en tipo TIMESTAMP*/
ALTER TABLE clientes ADD COLUMN fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP;

-- Otras opciones de tipo de dato y generar fecha y hora

/* DATETIME usa el mismo formato que TIMESTAMP, pero obtiene la hora del cliente (indicado en la proxima clase)
NOW() funcionalmente equivalente a CURRENT_TIMESTAMP */
ALTER TABLE clientes ADD COLUMN fecha_registro2 DATETIME DEFAULT NOW();

/* DATE solo obtiene la fecha
NOW() producira error sin los parentesis externos y una advertencia aun usandolos dado que no es el tipo
de formato de dato esperado 'YY-MM-DD HH:MM:SS' en lugar de 'YY-MM-DD'*/
ALTER TABLE clientes ADD COLUMN fecha_registro3 DATE DEFAULT (NOW());
