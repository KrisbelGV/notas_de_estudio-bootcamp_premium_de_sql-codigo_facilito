/* Contar caracteres de los campos campos descripcion_varchar y descripcion_text de la tabla productos
y ordenar los resultados descendentemente */
SELECT
    CHAR_LENGTH(descripcion_varchar) AS largo_varchar,
    CHAR_LENGTH(descripcion_text) AS largo_text
FROM productos
ORDER BY largo_varchar DESC;

/* Añadir CONSTRAINT CHECK sobre el campo precio en la tabla productos para que no admita valores inferiores a 0
Requiere un nombre, si no, el motor creara uno por defecto, mas no nos notificara cual de inmediato
Para obtenerlo deberemos consultar la estructura de la tabla u provocar su error */
ALTER TABLE productos
ADD CONSTRAINT chk_precio CHECK (precio > 0);

-- Actualizar campo precio en la tabla productos con un valor negativo (producira error dada la CONSTRAINT)
UPDATE productos
SET precio = -10.5;

/* Insertar valores aleatorios al campo precio de la tabla productos (como demostracion, luego deshacer cambios)
RAND() crea un numero al azar de punto florante entre 0 y 1 */
START TRANSACTION;
	UPDATE productos
	SET precio = RAND() * 20;
ROLLBACK;

/* Eliminar CONSTRAINT CHECK (no empleado en la clase)
ALTER TABLE productos
DROP CHECK chk_precio; */

/* Generar numeros aleatorios entre un rango:
FLOOR(RAND() * (maximo - minimo + 1) + minimo)
FLOOR () redondea al entero inferior mas cercano */

-- Ejemplo: Generar numeros aleatorios entre 10 y 5 y concatenar descripcion
SELECT CONCAT('Numero aleatorio entre 10 y 5: ', FLOOR(RAND() * (10 - 5 + 1) + 5)) AS numero_aleatorio;
