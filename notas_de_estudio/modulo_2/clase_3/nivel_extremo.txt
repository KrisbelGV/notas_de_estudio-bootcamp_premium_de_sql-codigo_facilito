/* Nivel Extremo: "Registro de Facturas"
Escenario: Tienes que diseñar la estructura de una tabla facturas para una empresa */


-- Seleccionar la BD correspondiente
USE tienda_db;


-- Crear la tabla facturas
CREATE TABLE facturas (
    factura_id INT AUTO_INCREMENT PRIMARY KEY,
    cliente_id INT,
    fecha_emision DATETIME DEFAULT NOW() CONSTRAINT chk_fecha_emision CHECK (fecha_emision > '1999-12-31 23:59:59'),
    monto_total DECIMAL(10,2) NOT NULL CONSTRAINT chk_monto_total CHECK (monto_total > 0), 
    estado VARCHAR(10) DEFAULT 'pendiente' CONSTRAINT chk_estado_valido CHECK (estado IN('pendiente', 'pagada', 'anulada')),
    FOREIGN KEY (cliente_id) REFERENCES clientes(id)
);


-- Insertar registros de ejemplo
INSERT INTO facturas(cliente_id, fecha_emision, monto_total)
VALUES (1, '2014-02-15 09:26:14', 100),
(2, '2021-11-09 13:54:01', 525.60);

/* Pero (debe cumplir) con estas reglas específicas:

Requisitos obligatorios:
1 Cada factura debe tener un número único (factura_id) */
SELECT factura_id FROM facturas;

-- 2 Se debe registrar la fecha de emisión (DEFAULT le asigna la actual)
INSERT INTO facturas(factura_id, cliente_id, monto_total)
VALUES (3, 3, 28.99);
SELECT factura_id, fecha_emision FROM facturas WHERE factura_id = 3;

-- 3 El monto total no puede ser negativo
INSERT INTO facturas(cliente_id, monto_total)
VALUES (2, -1);

-- 4 Solo puede haber tres estados posibles ("pendiente", "pagada", "anulada")
UPDATE facturas
SET estado = 'creada'
WHERE factura_id = 1;

-- 5 La factura debe estar asociada a un cliente (cliente_id)
SELECT cliente_id FROM facturas;

-- 6 Si no se especifica el estado, debe asumir "pendiente" por defecto
INSERT INTO facturas(factura_id, cliente_id, monto_total)
VALUES (4, 1, 78.6);
SELECT factura_id, estado FROM facturas WHERE factura_id = 4;
