-- 1 Obten los primeros 15 registros de la tabla film mostrando film_id, title y release_year
SELECT film_id, title, release_year FROM film LIMIT 15;

-- 2 Muestra todas las peliculas con duracion mayor a 120 minutos, mostrando film_id, title y length.
SELECT film_id, title, length FROM film WHERE length > 120;

-- 3 Lista todas las peliculas cuyo titulo contenga la palabra LOVE
SELECT film_id, title FROM film WHERE LOWER(title) LIKE '%love %' OR LOWER(title) LIKE '% love&' OR LOWER(title) LIKE '% love %';

SELECT film_id, title FROM film WHERE LOWER(title) REGEXP '[[:<:]]love[[:>:]]';
SELECT film_id, title FROM file WHERE LOWER(title) RLIKE '[[:<:]]love[[:>:]]';

-- 4 Obten las 10 peliculas mas recientes ordenadas por release_year de mayor a menor
SELECT title, release_year FROM film ORDER BY release_year DESC LIMIT 10;

-- 5 Cuenta cuantas peliculas existen por cada rating
SELECT COUNT(*), rating FROM film GROUP BY rating;

-- 6 Cuenta cuantas peliculas con duracion mayor a 100 minutos existen agrupadas por rating
SELECT COUNT(*), rating FROM film WHERE length > 100 GROUP BY rating;
