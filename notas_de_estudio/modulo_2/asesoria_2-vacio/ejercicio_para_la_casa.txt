/* Una empresa de tecnología ofrece soporte técnico a sus clientes a través de diferentes canales
(correo, chat, llamadas). Cada cliente puede tener múltiples tickets de soporte, los cuales son
atendidos por agentes. Los tickets pueden tener un estado (abierto, en progreso, cerrado) y estar
clasificados por categoría (técnico, facturación, funcionalidad, etc.). Un agente puede atender
varios tickets, y cada ticket puede registrar varias interacciones (respuestas o acciones del agente
o cliente). Además, cada agente pertenece a un equipo de soporte especializado.
Ejercicios de diseño de esquema.
	Identifica las entidades.
	Identifica las relaciones entre entidades.
	Crea tablas con sus columnas.
	Comparte la descripción de la tabla. */

-- Crear base de datos soporte_bd
CREATE DATABASE soporte_bd;

-- Seleccionar la base de datos creada
USE soporte_bd;


-- Crear tabla clientes
CREATE TABLE clientes(
	id INT AUTO_INCREMENT PRIMARY KEY,
    nombre_completo VARCHAR(200),
    email VARCHAR(320) NOT NULL, -- Limite según la RFC 5321, sección 4.5.3.1.3
    telefono VARCHAR(15) NOT NULL -- Maximo internacional (UIT-T, norma E.164)
);

-- Crear tabla agentes
CREATE TABLE agentes (
	id INT AUTO_INCREMENT PRIMARY KEY,
    tipo VARCHAR(6) CONSTRAINT tipo_valido CHECK (tipo IN('humano', 'bot')),
    equipo VARCHAR(100) NOT NULL
);

/* 
	Crear tabla tickets
	Posee una relacion N:N (dos llaves foraneas, una a 'clientes' y otra a 'agentes') 
*/
CREATE TABLE tickets (
	id INT AUTO_INCREMENT PRIMARY KEY, 
    cliente_id INT,
    agente_id INT,
    estado VARCHAR(11) CONSTRAINT estado_valido CHECK (estado IN('abierto', 'en progreso', 'cerrado')) DEFAULT 'abierto',
    categoria VARCHAR(13) CONSTRAINT categoria_valido CHECK (categoria IN('tecnico', 'facturacion', 'funcionalidad', 'otro')),
    FOREIGN KEY(cliente_id) REFERENCES clientes(id) ON DELETE CASCADE, 
    FOREIGN KEY(agente_id) REFERENCES agentes(id)
);

/* 
	Crear tabla interacciones
	Posee una relacion 1:N (una llave foranea a 'tickets')
*/
CREATE TABLE interacciones(
	id INT AUTO_INCREMENT PRIMARY KEY,
    ticket_id INT,
    accion VARCHAR(500),
    canal VARCHAR(8) CONSTRAINT canal_valido CHECK (canal IN('correo', 'chat', 'llamada')) NOT NULL,
    FOREIGN KEY(ticket_id) REFERENCES tickets(id) ON DELETE CASCADE
);
