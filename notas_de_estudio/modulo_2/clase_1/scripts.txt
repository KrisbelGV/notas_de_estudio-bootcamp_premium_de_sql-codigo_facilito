-- Ordenar los registros en base a las columnas categoria y precio de la tabla productos
SELECT *
FROM productos
ORDER BY categoria, precio DESC;

/* ORDER BY ordenara los campos en el orden especificado
Tomara en cuenta su criterio individual, ascendente por defecto
Siempre va al final de la consulta, solo antes de LIMIT o TOP */


-- Contar registros
SELECT COUNT(*) AS total
FROM productos;

-- Contar registros  agrupados por categoria
SELECT categoria, COUNT(*) AS total_registros
FROM productos
GROUP BY categoria;

-- Sumar precios agrupados por categoria
SELECT categoria, SUM(precio) AS suma_precio
FROM productos
GROUP BY categoria;

-- Calcular precio promedio agrupados por categoria
SELECT categoria, AVG(precio) AS promedio_precio
FROM productos
GROUP BY categoria;

/* Contar cantidad, sumar y calcular el precio promedio agrupados por categoria
y ordenar por suma_precio descendentemente. */
SELECT
	categoria, 
    COUNT(*) AS total_registros, 
    SUM(precio) AS suma_precio, 
    AVG(precio) AS promedio_precio      
FROM productos
GROUP BY categoria
ORDER BY suma_precio DESC;


/* Ejercicio practico 2: Agregar y ordenar datos
Reto: Escribe una consulta SQL que muestre las categorias con m�s de 2 productos,
ordenadas de mayor a menor cantidad. */
SELECT categoria, COUNT(*) AS total
FROM productos
GROUP BY categoria
HAVING COUNT(*) > 2
ORDER BY COUNT(*);
