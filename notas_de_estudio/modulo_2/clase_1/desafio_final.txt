/* Desafio final
Reto #2: Encuentra los clientes que han realizado más de 2 pedidos y muestra cuántos han hecho. */
SELECT nombre,
	(SELECT COUNT(*)
	FROM pedidos p
	GROUP BY cliente_id
	HAVING COUNT(*) > 2 AND c.id = p.cliente_id) AS numero_pedidos
FROM clientes c
WHERE id IN (SELECT pedidos.cliente_id
			FROM pedidos
            GROUP BY cliente_id
            HAVING COUNT(*) > 2);
