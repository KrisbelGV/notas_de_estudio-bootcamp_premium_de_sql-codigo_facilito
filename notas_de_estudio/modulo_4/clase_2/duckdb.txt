/* Ejercicio 1: Utillizando las tablas Books y Authors
escribir un LEFT JOIN que liste todos los autores y sus libros
inclusive si no escribieron ningun libro */
SELECT
    Name,
    Title
FROM Authors a
LEFT JOIN Books b ON a.AuthorID = b.AuthorID;

-- Sintaxis para seleccionar todo los campos en una tabla (desde DuckDB)
SELECT * FROM Publishers;


/* Ejercicio 2: Crear una query para la relacion Publishers
que liste el titulo del libro, autor y publisher */

-- Con INNER JOIN (solo aquellos que posean autor y publisher)
SELECT
    b.Title,
    a.Name,
    p.PublisherName
FROM Books b
INNER JOIN Authors a ON b.AuthorID = a.AuthorID
INNER JOIN Publishers p ON b.BookID = p.BookID;

-- Con LEFT JOIN (todos los libros)
SELECT
    b.Title,
    a.Name,
    p.PublisherName
FROM Books b
LEFT JOIN Authors a ON b.AuthorID = a.AuthorID
LEFT JOIN Publishers p ON b.BookID = p.BookID;


/* Conjunto de problemas
Pista: Usar UNION, INTERSECT y EXCEPT */

-- Problema 1: Combinar nombres unicos de authors y borrowers
SELECT name
FROM Authors
UNION
SELECT name
FROM Borrowers;

-- Problema 2: Mostrar nombres que aparecen en ambos
SELECT name
FROM Authors
INTERSECT
SELECT name
FROM Borrowers;

-- Problema 3: Indicar que authors no son borrowers
SELECT name
FROM Authors
EXCEPT
SELECT name
FROM Borrowers;


-- Operaciones anidadas

-- Obtener personas en tabla Authors cuyo nombre exista en la tabla Borrowers
SELECT *
FROM Authors
WHERE Name IN
	(SELECT Name FROM Borrowers);

/* Listar libros donde el id de su autor se encuentre en Authors
y que a su vez su nombre exista en Borrowers */
SELECT Title
FROM Books
WHERE AuthorID IN
	(SELECT AuthorID FROM Authors WHERE Name IN
		(SELECT Name FROM Borrowers));


-- Pregunta: cuales borrowers pidieron todos los libros (prestados en la tabla Borrowers)?

/* Con subconsultas (usando NOT EXISTS, operador diferencia)
Find borrowers who borrower every book in BookBorrowers */
SELECT DISTINCT outer_borrows.BorrowerID
FROM BookBorrowers outer_borrows
WHERE NOT EXISTS (
	-- List all unique BookIDs (our "complete set")
	SELECT BookID
    FROM (SELECT DISTINCT BookID FROM BookBorrowers) all_books
    WHERE NOT EXISTS (
		-- Check if this borrower has a record for each book
		SELECT *
        FROM BookBorrowers borrower_records
        WHERE borrower_records.BookID = all_books.BookID
        AND borrower_records.BorrowerID = outer_borrows.BorrowerID
    )
);
