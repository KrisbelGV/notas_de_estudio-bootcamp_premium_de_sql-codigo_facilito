-- Obtener la cantidad de productos por categoria (usando una tabla temporal intermedia)
CREATE TEMPORARY TABLE cantidad_total_categoria AS
SELECT
	categoria,
    COUNT(*) AS cantidad
FROM productos
GROUP BY categoria;

-- Obtener el numero de compras por categoria para cada cliente (usando una tabla temporal intermedia)
CREATE TEMPORARY TABLE compras_clientes_categoria AS
SELECT
	c.id_cliente,
	pr.categoria,
    COUNT(pr.categoria) AS cantidad
FROM clientes c
JOIN pedidos pe ON c.id_cliente = pe.id_cliente
JOIN detalles_pedido dp ON pe.id_pedido = dp.id_pedido
JOIN productos pr ON dp.id_producto = pr.id_producto
GROUP BY c.id_cliente, pr.categoria;

-- Mostrar el nombre de los clientes cuyo numero de compras en una categoria corresponda a su total
SELECT
	c.nombre,
	ccc.categoria
FROM clientes c
JOIN compras_clientes_categoria ccc ON c.id_cliente = ccc.id_cliente
JOIN cantidad_total_categoria ctc ON ccc.categoria = ctc.categoria
WHERE ctc.cantidad = ccc.cantidad;


-- 2 Mostrar la cantidad de ventas por mes en 2024

-- Crear tabla temporal con las compras por cada mes
CREATE TEMPORARY TABLE compras_por_mes AS
SELECT
    MONTH(fecha_pedido) AS mes,
    COUNT(*) AS cantidad
FROM pedidos
WHERE fecha_pedido BETWEEN '2024-01-01' AND '2024-12-31'
GROUP BY mes;

-- Generar fechas entre 2024-01-01 hasta 2024-12-01
WITH RECURSIVE meses_2024 AS (
	SELECT DATE('2024-01-01') AS fecha
	UNION ALL
	SELECT DATE_ADD(fecha, INTERVAL 1 MONTH)
	FROM meses_2024 WHERE fecha < '2024-12-01'
)

-- Unir fechas con su cantidad de ventas y llenar campos faltantes
SELECT
	DATE_FORMAT(m.fecha, "%Y-%m") AS mes,
    COALESCE(cantidad, 0) AS cantidad
FROM meses_2024 m
LEFT JOIN compras_por_mes cpm ON MONTH(m.fecha) = cpm.mes;
