-- Calcular pedidos realizados en febrero de 2024

-- Con BETWEEN
SELECT * FROM pedidos
WHERE fecha_pedido BETWEEN '2024-02-01' AND '2024-02-29';

-- Con igual o mayor/menor
SELECT * FROM pedidos
WHERE fecha_pedido >= '2024-02-01' AND fecha_pedido <= '2024-02-29';

/* Ambas consultas son validas
Se usa una u otra a convenir (por rendimiento) */


-- Extraer año, mes y dia (numerico y nombre) de una fecha
SELECT fecha_pedido,
	YEAR(fecha_pedido) AS anio,
    MONTH(fecha_pedido) AS mes,
    DAY(fecha_pedido) AS dia,
    MONTHNAME(fecha_pedido) AS nombre_mes,
    DAYNAME(fecha_pedido) AS nombre_dia
FROM pedidos;


-- Restar fechas
SELECT DATEDIFF(CURDATE(), '2025-03-01') AS resta;

-- Agregar dias
SELECT DATE_ADD(CURDATE(), INTERVAL 5 DAY) AS suma;


/* COMMON TABLE EXPRESSION (expresion de tabla comun)
Es un conjunto de resultados de uso temporal */

-- Generar fechas entre 2024-01-01 hasta 2024-12-01
WITH RECURSIVE meses AS (
	SELECT DATE('2024-01-01') AS fecha
	UNION ALL
	SELECT DATE_ADD(fecha, INTERVAL 1 MONTH)
	FROM meses WHERE fecha < '2024-12-01'
)
SELECT * FROM meses;


/* Calcular acumulados mes a mes
OVER aplica la funcion iteradamente
ORDER BY tras FROM solo al final */
SELECT
	fecha_pedido,
    total,
    SUM(total) OVER (ORDER BY fecha_pedido) AS total_acumulado
FROM pedidos; 


-- Calcular clientes que hicieron compras en los ultimos 3 meses

-- Con subconsultas
SELECT nombre
FROM clientes
WHERE id_cliente IN (
	SELECT id_cliente
    FROM pedidos
    WHERE fecha_pedido >= CURDATE() - INTERVAL 3 MONTH
);

-- Con JOIN
SELECT c.nombre
FROM clientes c
JOIN pedidos p ON c.id_cliente = p.id_cliente
WHERE fecha_pedido >= CURDATE() - INTERVAL 3 MONTH
GROUP BY c.nombre;


-- Ejercicio: Mostrar productos que no han sido vendidos en los ultimos 6 meses
SELECT pr.nombre
FROM productos pr
LEFT JOIN detalles_pedido dp ON pr.id_producto = dp.id_producto
JOIN pedidos pe ON dp.id_pedido = pe.id_pedido
WHERE pe.fecha_pedido < CURDATE() - INTERVAL 6 MONTH;


-- Obtener pedidos con detalle en marzo de 2024
SELECT
	c.nombre,
    p.nombre AS producto,
    pe.fecha_pedido
FROM clientes c
JOIN pedidos pe ON c.id_cliente = pe.id_cliente
JOIN detalles_pedido dp ON pe.id_pedido = dp.id_pedido
JOIN productos p ON dp.id_producto = p.id_producto
WHERE MONTH(pe.fecha_pedido) = 3 AND YEAR(pe.fecha_pedido) = 2024;

/* Procedimiento almacenado que genera fechas en un rango
Emplea una sentencia preparada (no visto en la clase) */
DELIMITER //
CREATE PROCEDURE GENERATE_SERIES(fecha_inicio DATE, fecha_fin DATE, intervalo VARCHAR(50)) -- cantidad INT, unidad VARCHAR(10))
BEGIN 
	SET @sp = CONCAT('
	WITH RECURSIVE fechas AS (
		SELECT''', fecha_inicio,''' AS fecha
		UNION ALL
		SELECT DATE_ADD(fecha, INTERVAL ', intervalo,')
		FROM fechas WHERE fecha <''', fecha_fin,'''
	)
    SELECT fecha FROM fechas');
    
    PREPARE stmt FROM @sp;
	EXECUTE stmt;
	DEALLOCATE PREPARE stmt;
END//
DELIMITER ;

-- Obtener fechas de 2024-01-01 a 2024-12-31 con intervalos de 1 mes
CALL GENERATE_SERIES('2024-01-01', '2024-12-31', '1 MONTH');
