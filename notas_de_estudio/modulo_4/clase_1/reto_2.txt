-- Calcular horas trabajadas por dia con la entrada y salida de un empleado

/* Asumimos que cada asistencia posee su hora de entrada y salida
Ademas que fueron registradas de forma consecutiva
Obviaremos trabajar con el dia */

-- Seleccionar la BD para este ejercicio
USE control_asistencia;

-- Crear tabla temporal con las horas de entrada
CREATE TEMPORARY TABLE horas_entrada AS
SELECT
	id_registro,
    id_empleado,
	hora
FROM asistencia
WHERE tipo = "entrada";

-- Crear tabla temporal con las horas de entrada
CREATE TEMPORARY TABLE horas_salida AS
SELECT
	id_registro,
    id_empleado,
	hora
FROM asistencia
WHERE tipo = "salida";

-- Emparejar horas de entrada y salida para cada empleado, sumar su tiempo
SELECT
	e.nombre,
	CONCAT(FORMAT(SUM(DATE_FORMAT(TIMEDIFF(hs.hora, he.hora), '%H.%i')), 2), " h.m") AS horas_totales
FROM horas_entrada he
JOIN empleados e ON he.id_empleado = e.id_empleado
JOIN horas_salida hs ON (he.id_registro + 1) = hs.id_registro
GROUP BY he.id_empleado;
