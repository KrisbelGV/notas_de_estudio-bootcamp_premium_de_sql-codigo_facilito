/* Mini proyecto: Modelando BD con varias relaciones.

Escenario:
Vamos a diseñar una base de datos para gestionar préstamos de biblioteca.

Tablas:
usuarios(id, nombre, email)
libros(id, titulo, autor, año_publicación)
prestamos(id, usuario_id, libro_id, fecha_prestamo, fecha_devolucion)

Objetivo:
Relacionar correctamente prestamos con usuarios y libros usando FOREIGN KEYS
Definir ON DELETE CASCADE en usuario_id. */


-- Crear BD Biblioteca
CREATE DATABASE biblioteca_db;

-- Seleccionar la BD creada
USE biblioteca_db;


-- Crear tabla usuarios
CREATE TABLE usuarios(
    usuario_id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE
);

-- Crear tabla libros
CREATE TABLE libros (
    libro_id INT AUTO_INCREMENT PRIMARY KEY,
    titulo VARCHAR(100),
    autor VARCHAR(100),
    año_publicacion DATE
);

-- Crear tabla pedidos
CREATE TABLE pedidos (
    pedido_id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT,
    libro_id INT,
    fecha_prestamo DATE DEFAULT (CURDATE()),
    fecha_devolucion DATE DEFAULT ((CURDATE()) + INTERVAL 6 MONTH), -- Fecha actual sumado un mes (visto en "Condiciones de consulta", clase #1 modulo 4)
    CONSTRAINT chk_fecha_prestamo_entrega_coincidientes CHECK (fecha_devolucion = fecha_prestamo + INTERVAL 6 MONTH), -- Verificar plazo valido (p. ej. 6 meses)
    FOREIGN KEY(usuario_id) REFERENCES usuarios(usuario_id) ON DELETE CASCADE,
    FOREIGN KEY(libro_id) REFERENCES libros(libro_id)
);
