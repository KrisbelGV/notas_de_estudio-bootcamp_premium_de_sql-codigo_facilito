-- Ejecutar en la bd de la asesoria anterior


-- Obtener el total de las ordenes por usuario

-- Con GROUP BY (solo el user_id y la suma total)
SELECT
    orders.user_id,
    SUM(orders.total)
FROM orders
GROUP BY orders.user_id;

-- Con OVER + PARTITION (aun es posible acceder al resto de campos)
SELECT
    orders.user_id,
	orders.id AS order_id,
    SUM(orders.total) OVER(
		PARTITION BY orders.user_id
    ) AS total
FROM orders;


-- Sumar el total de las ordenes por usuario ordenadas por su estatus
SELECT
    orders.user_id,
    orders.status,
    SUM(orders.total) OVER(
		PARTITION BY orders.user_id ORDER BY orders.status
    ) AS total
FROM orders;

-- Sumar el total de las ordenes por usuario ordenadas por su total
SELECT
    orders.user_id,
    orders.status,
    SUM(orders.total) OVER(
		PARTITION BY orders.user_id ORDER BY orders.total
    ) AS total
FROM orders;


-- Obtener el nombre del usuario, el total de sus ordenes y un mensaje de cupon de poseerlo
SELECT
    users.name,
    SUM(orders.total),
    IF(SUM(orders.total) >= 500, "si", "no") AS posee_cupon,
    CASE
		WHEN SUM(orders.total) >= 1000 THEN "Supercupon"
        WHEN SUM(orders.total) >= 500 THEN "Cupon de descuento"
        ELSE ""
    END AS menssaje
FROM orders
INNER JOIN users ON orders.user_id = users.id
GROUP BY orders.user_id;


-- Desactivar historial de tiempos de ejecucion
SET PROFILING = 0;
